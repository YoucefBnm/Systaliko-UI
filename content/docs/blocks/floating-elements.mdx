## Usage

```tsx
<FloatingElements className="bg-secondary flex-1">
  {ELEMENTS.map((el) => (
    <InfiniteFloatingItem
      className={`absolute rounded ring ring-ring/20 shadow-2xs size-20 ${el.className}`}
      key={el.id}
      depth={depth}
      amplitude={amplitude}
      speed={speed}
      phase={el.phase}
      stiffness={stiffness}
      damping={damping}
    />
  ))}
</FloatingElements>
```
