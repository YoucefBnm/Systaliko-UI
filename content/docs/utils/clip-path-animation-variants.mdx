## Usage

```tsx
const clipPathVariants = clip_path_animation_variants['curtain-down'];
<motion.div
  variants={clipPathVariants}
  transition={{ duration: 0.3, delay: 0.2, ease: 'easeInOut' }}
  initial="hidden"
  whileInView="visible"
  viewport={{ once: true }}
/>;
```
